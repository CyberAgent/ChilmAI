17def numbered_pattern(template: str) -> re.Pattern[str]: 

18 """列名テンプレート内の数字部分を取り出す正規表現を作る。 

19 

20 小文字 ASCII が直後に続かない大文字 `N` を、数字の位置として扱う。 

21 該当する `N` がない場合は、テンプレート末尾に数字が続くものとして扱う。 

22 複数の `N` が該当する場合は、最初の `N` だけを使う。 

23 `No.` のように `N` の直後が小文字の場合はプレースホルダではなく文字として扱う。 

24 `NAME_` のように大文字 ASCII と隣接する `N` はプレースホルダ扱いになるため、 

25 そのようなテンプレートは避ける。 

26 

27 例: 

28 "N歳定員" → ^(\\d+)歳定員$ は 0歳定員, 1歳定員 ... に一致 

29 "定員N" → ^定員(\\d+)$ は 定員0, 定員1 ... に一致 

30 "No." → ^No\\.(\\d+)$ は No.1, No.2 ... に一致（N は文字扱い） 

31 "capacity_age" → ^capacity_age(\\d+)$ 

32 """ 

33 m = re.search(r"N(?![a-z])", template) 

34 if m: 

35 idx = m.start() 

36 before = re.escape(template[:idx]) 

37 after = re.escape(template[idx + 1 :]) 

38 return re.compile(rf"^{before}(\d+){after}$", re.ASCII) 

39 return re.compile(rf"^{re.escape(template)}(\d+)$", re.ASCII)

42class InputParser: 

43 """アップロードされたファイルを正規化済みの `pandas.DataFrame` に変換する。 

44 

45 CSV と Excel を受け付け、全角を含む列名を NFKC で正規化し、 

46 利用者が設定した列名を `child_id`, `pref_1`, `capacity_age0` などの 

47 ChilmAI 内部列名へ変換する。 

48 """

50 @staticmethod 

51 def _normalize_file_format(file_format: str) -> FileFormat: 

52 normalized = file_format.lower().strip() 

53 if normalized == "csv": 

54 return "csv" 

55 if normalized in {"excel", "xlsx", "xls", "xlsm", "xlsb"}: 

56 return "excel" 

57 raise ChilmError(f"Unsupported file format: {file_format}", code=ErrorCode.UNSUPPORTED_FILE_FORMAT) 

58 

59 @staticmethod 

60 def _read_csv(source: IO[bytes] | BytesIO, **kwargs) -> pd.DataFrame: 

61 """CSV を読み込む。UTF-8 以外の文字コードは分かりやすいエラーへ変換する。 

62 

63 Excel の既定の「CSV (コンマ区切り)」保存は Shift-JIS になるため、UTF-8 前提の 

64 `pd.read_csv` が `UnicodeDecodeError` を送出する。この生の英語例外をそのまま画面に 

65 出すと利用者が原因を特定できないため、エラーコード付きの日本語メッセージ 

66 （ErrorCode.CSV_ENCODING_ERROR）に変換する。 

67 """ 

68 try: 

69 return pd.read_csv(source, **kwargs) 

70 except UnicodeDecodeError as exc: 

71 raise ChilmError( 

72 "ファイルの文字コードが UTF-8 ではないため読み込めませんでした。" 

73 "Excel の「名前を付けて保存」で「CSV UTF-8 (コンマ区切り)」形式を選んで保存し直すか、" 

74 "Excel 形式（.xlsx）のままアップロードしてください。", 

75 code=ErrorCode.CSV_ENCODING_ERROR, 

76 ) from exc

78 @classmethod 

79 def _read_dataframe(cls, file_bytes: bytes, file_format: str) -> pd.DataFrame: 

80 normalized = cls._normalize_file_format(file_format) 

81 buffer = BytesIO(file_bytes) 

82 if normalized == "csv": 

83 return cls._read_csv(buffer, dtype=str) 

84 # calamine (MIT) を明示。pandas は .xlsb の既定エンジンとして LGPLv3 の 

85 # pyxlsb を選ぶが、これを廃し xlsx/xls/xlsm/xlsb を単一エンジンで読む。 

86 # engine を省くと pyxlsb 不在時に xlsb で ImportError になる。 

87 return pd.read_excel(buffer, sheet_name=0, dtype=str, engine="calamine")

89 @classmethod 

90 def read_columns(cls, fileobj: IO[bytes], file_format: str) -> list[str]: 

91 normalized = cls._normalize_file_format(file_format) 

92 if normalized == "csv": 

93 df = cls._read_csv(fileobj, dtype=str, nrows=0) 

94 else: 

95 df = pd.read_excel(fileobj, sheet_name=0, dtype=str, nrows=0, engine="calamine") 

96 return [unicodedata.normalize("NFKC", str(c)) for c in df.columns if c is not None]

98 @staticmethod 

99 def _apply_mapping(df: pd.DataFrame, mapping: dict[str, str]) -> pd.DataFrame: 

100 _PREFIX_KEYS = {"preference_prefix", "score_prefix", "capacity_prefix"} 

101 

102 # Normalize column names: full-width digits/letters → half-width (NFKC). 

103 # str(col) handles non-string columns (numeric, datetime) from Excel files. 

104 norm_map = {col: unicodedata.normalize("NFKC", str(col)) for col in df.columns} 

105 seen: set[str] = set() 

106 duplicates: list[str] = [] 

107 for normalized in norm_map.values(): 

108 if normalized in seen: 

109 duplicates.append(normalized) 

110 seen.add(normalized) 

111 if duplicates: 

112 raise ChilmError( 

113 f"正規化後に列名が重複します（全角・半角の混在を確認してください）: {duplicates}", 

114 code=ErrorCode.DUPLICATE_COLUMN_AFTER_NORM, 

115 ) 

116 df = df.rename(columns=norm_map) 

117 

118 rename_map: dict[str, str] = {} 

119 for internal_name, user_name in mapping.items(): 

120 if internal_name in _PREFIX_KEYS: 

121 continue 

122 normalized_user_name = ( 

123 unicodedata.normalize("NFKC", str(user_name)) if user_name is not None else user_name 

124 ) 

125 if normalized_user_name in df.columns: 

126 rename_map[normalized_user_name] = internal_name 

127 

128 mapped = df.rename(columns=rename_map) 

129 

130 # Convert preference columns (e.g. 希望保育園ID_1) to pref_1, pref_2 ... 

131 pref_pattern = numbered_pattern( 

132 unicodedata.normalize("NFKC", mapping.get("preference_prefix", "pref_")) 

133 ) 

134 for col in list(mapped.columns): 

135 if not isinstance(col, str): 

136 continue 

137 if re.fullmatch(r"pref_\d+", col, re.ASCII): 

138 continue 

139 m = pref_pattern.fullmatch(col) 

140 if m: 

141 mapped = mapped.rename(columns={col: f"pref_{m.group(1)}"}) 

142 

143 # Convert score columns (e.g. スコア1) to score_1, score_2 ... 

144 score_pattern = numbered_pattern( 

145 unicodedata.normalize("NFKC", mapping.get("score_prefix", "score_")) 

146 ) 

147 for col in list(mapped.columns): 

148 if not isinstance(col, str): 

149 continue 

150 if re.fullmatch(r"score_\d+", col, re.ASCII): 

151 continue 

152 m = score_pattern.fullmatch(col) 

153 if m: 

154 mapped = mapped.rename(columns={col: f"score_{m.group(1)}"}) 

155 

156 # Allow bare "score" column (no suffix) as an alias for "score_1". 

157 if "score" in mapped.columns: 

158 score_numbered = [ 

159 c for c in mapped.columns if isinstance(c, str) and re.fullmatch(r"score_\d+", c) 

160 ] 

161 if score_numbered: 

162 if "score_1" in score_numbered: 

163 raise ChilmError( 

164 "「score」列と「score_1」列が両方存在します。どちらか一方だけを使用してください。", 

165 code=ErrorCode.SCORE_COLUMN_CONFLICT_1, 

166 ) 

167 raise ChilmError( 

168 f"「score」列と番号付きスコア列（{score_numbered[0]} など）が混在しています。" 

169 "「score_1」から始まる列名を使用してください。", 

170 code=ErrorCode.SCORE_COLUMN_CONFLICT_2, 

171 ) 

172 mapped = mapped.rename(columns={"score": "score_1"}) 

173 

174 # 整数を表す '.0' サフィックスを除去する（pandas が float→str 化した値や 

175 # Excel/CSV の生データに '512200400.0' のような値が含まれるケースを救済）。 

176 # 整数値以外の小数（'100.5' 等）は validator が後段で弾けるよう手を加えない。 

177 _DOT_ZERO_RE = re.compile(r"^(-?\d+)\.0+$") 

178 for col in list(mapped.columns): 

179 if isinstance(col, str) and re.fullmatch(r"score_\d+", col, re.ASCII): 

180 mapped[col] = mapped[col].map( 

181 lambda v: _DOT_ZERO_RE.sub(r"\1", str(v).strip()) if pd.notna(v) else v 

182 ) 

183 

184 # Convert capacity columns (e.g. N歳定員, 定員N) to capacity_age0, capacity_age1 ... 

185 capacity_pattern = numbered_pattern( 

186 unicodedata.normalize("NFKC", mapping.get("capacity_prefix", "capacity_age")) 

187 ) 

188 for col in list(mapped.columns): 

189 if not isinstance(col, str): 

190 continue 

191 if re.fullmatch(r"capacity_age\d+", col, re.ASCII): 

192 continue 

193 m = capacity_pattern.fullmatch(col) 

194 if m: 

195 mapped = mapped.rename(columns={col: f"capacity_age{m.group(1)}"}) 

196 

197 return mapped

225 def parse_combination( 

226 self, 

227 *, 

228 file_bytes: bytes, 

229 file_format: str, 

230 mapping: dict[str, str], 

231 ) -> pd.DataFrame: 

232 """組み合わせファイルを読み込み、内部列名へ変換して返す。 

233 

234 Returns: 

235 以下の内部列を持つ DataFrame: 

236 

237 - household_id (str): ファミリーコード 

238 - rank (str): 総当たり順位 

239 - child_code_0 (str | float): 宛名コード1（0-indexed）。列が存在しない行は NaN。 

240 - child_code_1 (str | float): 宛名コード2（0-indexed）、以降同様。列が存在しない行は NaN。 

241 - facility_0 (str | float): 希望施設1（0-indexed）。列が存在しない行は NaN。 

242 - facility_1 (str | float): 希望施設2（0-indexed）、以降同様。列が存在しない行は NaN。 

243 """ 

244 df = self._read_dataframe(file_bytes, file_format) 

245 

246 # NFKC 正規化 

247 norm_cols = {col: unicodedata.normalize("NFKC", str(col)) for col in df.columns} 

248 seen: set[str] = set() 

249 duplicates: list[str] = [] 

250 for normalized in norm_cols.values(): 

251 if normalized in seen: 

252 duplicates.append(normalized) 

253 seen.add(normalized) 

254 if duplicates: 

255 raise ChilmError( 

256 f"正規化後に列名が重複します: {duplicates}", 

257 code=ErrorCode.DUPLICATE_COLUMN_AFTER_NORM, 

258 ) 

259 df = df.rename(columns=norm_cols) 

260 

261 # 単一列のリネーム（household_id, rank） 

262 rename_map: dict[str, str] = {} 

263 for internal, user in ( 

264 ("household_id", mapping.get("household_id", "ファミリーコード")), 

265 ("rank", mapping.get("rank", "総当たり順位")), 

266 ): 

267 normalized_user = unicodedata.normalize("NFKC", str(user)) if user else "" 

268 if normalized_user in df.columns: 

269 rename_map[normalized_user] = internal 

270 df = df.rename(columns=rename_map) 

271 

272 # child_code_prefix → child_code_0, child_code_1, ... 

273 child_tmpl = unicodedata.normalize("NFKC", mapping.get("child_code_prefix", "宛名コードN")) 

274 child_pat = numbered_pattern(child_tmpl) 

275 for col in list(df.columns): 

276 if not isinstance(col, str): 

277 continue 

278 if re.fullmatch(r"child_code_\d+", col, re.ASCII): 

279 continue 

280 m = child_pat.fullmatch(col) 

281 if m: 

282 num = int(m.group(1)) 

283 if num < 1: 

284 continue # 0始まり列は不正（負の内部列名を生成しない） 

285 df = df.rename(columns={col: f"child_code_{num - 1}"}) 

286 

287 # facility_prefix → facility_0, facility_1, ... 

288 facility_tmpl = unicodedata.normalize("NFKC", mapping.get("facility_prefix", "希望施設N")) 

289 facility_pat = numbered_pattern(facility_tmpl) 

290 for col in list(df.columns): 

291 if not isinstance(col, str): 

292 continue 

293 if re.fullmatch(r"facility_\d+", col, re.ASCII): 

294 continue 

295 m = facility_pat.fullmatch(col) 

296 if m: 

297 num = int(m.group(1)) 

298 if num < 1: 

299 continue # 0始まり列は不正（負の内部列名を生成しない） 

300 df = df.rename(columns={col: f"facility_{num - 1}"}) 

301 

302 return df
