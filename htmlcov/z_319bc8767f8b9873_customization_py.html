1"""リファレンスアプリ（Web UI・サーバ API）の差し替え口。 

2 

3自治体ごとに異なる前処理・入力ファイルの読み込み・結果 Excel・画面の入力項目を、 

4``apps/`` を改変せずに差し替えるための仕組み。``WebCustomization`` を継承した 

5クラスを ``apps/`` の外に実装し、環境変数 ``CHILMAI_WEB_CUSTOMIZATION`` に 

6``module:ClassName`` 形式で指定する:: 

7 

8 CHILMAI_WEB_CUSTOMIZATION=mytown.web:MyTownCustomization \\ 

9 uv run uvicorn apps.api.main:app --host 127.0.0.1 --port 8501 

10 

11配布 ZIP（``apps/packager.py``）をこの環境変数を設定してビルドすると、指定した 

12モジュールが exe に組み込まれ、起動時に環境変数がなくても同じカスタマイズを使う。 

13 

14実装例は ``examples/sample_web_customization.py`` を参照。 

15"""

46@dataclass(frozen=True) 

47class ParamField: 

48 """Web UI のアップロード画面に追加する入力項目。 

49 

50 入力値は文字列として ``WebCustomization`` の各メソッドに ``params`` で渡される。 

51 ``checkbox`` はチェックありで ``"true"``、なしで ``"false"`` になる。 

52 サーバ API で項目そのものを省略した場合は ``default`` を使う。空欄で送られた場合は 

53 空文字のまま扱い、``required`` の項目ではエラーになる。 

54 

55 Attributes: 

56 name: 項目名。英数字と ``_`` のみ。``params`` のキーになる。 

57 label: 画面に表示する項目名。 

58 type: 入力の種類。``"text"``・``"number"``・``"select"``・``"checkbox"``。 

59 choices: ``select`` の選択肢。 

60 default: 画面の初期値。サーバ API で項目を省略した場合もこの値を使う。 

61 ``checkbox`` では ``"true"`` か ``"false"``。 

62 required: 入力必須かどうか。 

63 help_text: 入力欄の下に表示する説明。 

64 """ 

65 

66 name: str 

67 label: str 

68 type: Literal["text", "number", "select", "checkbox"] = "text" 

69 choices: tuple[str, ...] = () 

70 default: str = "" 

71 required: bool = False 

72 help_text: str = ""

128def build_default_workbook( 

129 result: dict[str, Any], 

130 meta: Mapping[str, str] | None = None, 

131 params: Mapping[str, str] | None = None, 

132) -> openpyxl.Workbook: 

133 """標準の結果 Excel（``result``・``meta`` の 2 シート）を作成する。 

134 

135 ``WebCustomization.build_result_workbook`` で、この結果にシートや書式を 

136 追加する場合にも使える。 

137 """ 

138 output_columns: list[str] = result.get("output_columns", []) 

139 output_rows: list[dict[str, Any]] = result.get("output_rows", []) 

140 

141 wb = openpyxl.Workbook() 

142 ws = wb.active 

143 ws.title = "result" 

144 ws.append(output_columns) 

145 for row in output_rows: 

146 ws.append([row.get(col, "") for col in output_columns]) 

147 

148 meta_ws = wb.create_sheet(title="meta") 

149 meta_ws.append(["key", "value"]) 

150 meta_items: list[tuple[str, str]] = [ 

151 ("schema_version", str(SCHEMA_VERSION)), 

152 ("active_profile", (meta or {}).get("active_profile", "")), 

153 ("executed_at", (meta or {}).get("executed_at", "")), 

154 ] 

155 meta_items.extend((f"param.{name}", value) for name, value in (params or {}).items()) 

156 for key, value in meta_items: 

157 meta_ws.append([key, value]) 

158 return wb

161def neutralize_formulas(wb: openpyxl.Workbook) -> None: 

162 """すべてのシートの数式セルを文字列セルに変える。

165 Excel で数式として実行されないよう、文字列型に戻す。

167 for ws in wb.worksheets:

178def parse_params(fields: Sequence[ParamField], raw: Mapping[str, Any]) -> dict[str, str]: 

179 """入力値を検証し、``fields`` の定義どおりの ``{name: 値}`` に整える。 

180 

181 Raises: 

182 ChilmError: 定義にない項目がある場合、必須項目が未入力の場合、 

183 選択肢にない値や数値でない値、checkbox に真偽値でない値が入力された場合。 

184 """ 

185 known = {f.name for f in fields} 

186 unknown = sorted(str(key) for key in raw if key not in known) 

187 if unknown: 

188 raise _invalid_param(f"定義されていない入力項目が指定されました: {unknown}") 

189 

190 params: dict[str, str] = {} 

191 for field in fields: 

192 value = raw.get(field.name) 

193 if field.type == "checkbox": 

194 if value is None: 

195 text = field.default or "false" 

196 else: 

197 normalized = str(value).strip().lower() 

198 if normalized in _CHECKBOX_TRUE: 

199 text = "true" 

200 elif normalized in _CHECKBOX_FALSE: 

201 text = "false" 

202 else: 

203 raise _invalid_param( 

204 f"「{field.label}」には true か false を指定してください: {value!r}" 

205 ) 

206 params[field.name] = text 

207 continue 

208 # 項目の省略は既定値、空欄での送信は「未入力」として扱う。 

209 # 任意の select で「—」を選んだ場合などに、既定値で上書きしないため。 

210 text = field.default if value is None else str(value).strip() 

211 if not text: 

212 if field.required: 

213 raise _invalid_param(f"「{field.label}」を入力してください。") 

214 params[field.name] = "" 

215 continue 

216 if field.type == "select" and text not in field.choices: 

217 raise _invalid_param(f"「{field.label}」には選択肢の中から指定してください: {text!r}") 

218 if field.type == "number" and not _NUMBER_RE.fullmatch(text): 

219 raise _invalid_param(f"「{field.label}」には半角の数値を入力してください: {text!r}") 

220 params[field.name] = text 

221 return params

224def _check_fields(fields: Sequence[ParamField]) -> None: 

225 seen: set[str] = set() 

226 for field in fields: 

227 if not _PARAM_NAME_RE.fullmatch(field.name): 

228 raise ValueError(f"入力項目の name には英数字と _ のみ使えます: {field.name!r}") 

229 if field.name in seen: 

230 raise ValueError(f"入力項目の name が重複しています: {field.name!r}") 

231 seen.add(field.name) 

232 if field.type not in _FIELD_TYPES: 

233 raise ValueError(f"入力項目の type が正しくありません: {field.name!r} ({field.type!r})") 

234 if field.type == "checkbox" and field.default not in ("", "true", "false"): 

235 raise ValueError(f'checkbox の default は "true" か "false" で指定してください: {field.name!r}') 

236 if field.type == "number" and field.default and not _NUMBER_RE.fullmatch(field.default): 

237 raise ValueError(f"number の default が半角の数値ではありません: {field.name!r}") 

238 if field.type == "select": 

239 if not field.choices: 

240 raise ValueError(f"select の入力項目には choices が必要です: {field.name!r}") 

241 if field.default and field.default not in field.choices: 

242 raise ValueError(f"default が choices に含まれていません: {field.name!r}")

245def load_customization(spec: str | None = None) -> WebCustomization: 

246 """``module:ClassName`` 形式の指定からカスタマイズを読み込む。 

247 

248 ``spec`` を省略した場合は環境変数 ``CHILMAI_WEB_CUSTOMIZATION`` を使う。 

249 環境変数も未設定の場合は、配布 ZIP のビルド時に同梱した指定（``BUNDLED_SPEC_FILE``）を 

250 使い、それもない場合は標準の ``WebCustomization`` を返す。 

251 指定に誤りがある場合は、標準の挙動に切り替えずに例外を送出する。 

252 """ 

253 if spec is None: 

254 spec = os.getenv(ENV_VAR, "") 

255 if not spec.strip() and BUNDLED_SPEC_FILE.is_file(): 

256 spec = BUNDLED_SPEC_FILE.read_text(encoding="utf-8") 

257 spec = spec.strip() 

258 if not spec: 

259 return WebCustomization() 

260 

261 module_name, sep, class_name = spec.partition(":") 

262 if not sep or not module_name or not class_name: 

263 raise ValueError(f"{ENV_VAR} は module:ClassName の形式で指定してください: {spec!r}") 

264 module = importlib.import_module(module_name) 

265 cls = getattr(module, class_name, None) 

266 if not (isinstance(cls, type) and issubclass(cls, WebCustomization)): 

267 raise TypeError(f"{spec!r} は WebCustomization のサブクラスではありません。") 

268 customization = cls() 

269 _check_fields(customization.fields) 

270 return customization
