17def _resolve_original_col(col: str, df: pd.DataFrame) -> str: 

18 """NFKC 正規化後に `col` と一致する、df 上の実際の列名を返す。 

19 

20 設定には read_columns() が表示した NFKC 正規化済みの列名が保存される一方、 

21 read_raw() は元ファイルの全角文字を保持するため、この解決が必要になる。 

22 """ 

23 if col in df.columns: 

24 return col 

25 norm_col = unicodedata.normalize("NFKC", col) 

26 for actual in df.columns: 

27 if unicodedata.normalize("NFKC", str(actual)) == norm_col: 

28 return actual 

29 return col

32class MatchingService: 

33 """パース、検証、マッチング、結果整形をまとめて扱う。 

34 

35 FastAPI の HTTP レイヤーを経由せずに ChilmAI を実行したいアプリケーション向けの 

36 推奨 Python API エントリポイント:: 

37 

38 from chilmai.generic.config import DEFAULT_CONFIG 

39 from chilmai.generic.service import MatchingService 

40 

41 service = MatchingService() 

42 result = service.match( 

43 children_file_bytes=children_bytes, 

44 children_file_format="csv", 

45 daycares_file_bytes=daycares_bytes, 

46 daycares_file_format="csv", 

47 mapping=DEFAULT_CONFIG,

49

51 preprocessor: 自治体カスタム前処理。``None`` の場合はすべてパススルーの 

52 ``BasePreprocessor`` を使う。前処理が呼ばれる順序は 

53 ``BasePreprocessor`` を参照。 

54 """

62 def _parse_combination(

71 

72 Args:

74 combination_file_format: ``"csv"`` または ``"xlsx"``。 

75 mapping: ``"combination"``・``"children"`` キーを含む列名マッピング。 

76 children_df: 相互参照の検証に使う申込者データ（内部列名）。 

77 daycares_df: 相互参照の検証に使う保育所データ（内部列名）。 

78 

79 Returns: 

80 成功時は ``(組み合わせ DataFrame, [])``、

96 children_mapping=mapping.get("children", {}), 

97 )

102 def validate( 

103 self, 

104 *, 

105 children_file_bytes: bytes, 

106 children_file_format: str, 

107 daycares_file_bytes: bytes, 

108 daycares_file_format: str, 

109 mapping: dict[str, dict[str, str]], 

110 combination_file_bytes: bytes | None = None, 

111 combination_file_format: str | None = None, 

112 ) -> dict[str, Any]: 

113 """アップロードされた申込者ファイルと保育所ファイルを検証する。 

114 

115 マッチングは実行せず、パース → 自治体カスタムバリデーション → 

116 ChilmAI 汎用バリデーションまでを行う。公開 ``/validate`` HTTP エンドポイントと 

117 同じ構造を返す。すべてキーワード引数で渡す。 

118 

119 Args: 

120 children_file_bytes: 申込者ファイルの内容。 

121 children_file_format: 申込者ファイルの形式。``"csv"`` または ``"xlsx"``。 

122 daycares_file_bytes: 保育所ファイルの内容。 

123 daycares_file_format: 保育所ファイルの形式。``"csv"`` または ``"xlsx"``。 

124 mapping: 列名マッピング。``"children"``・``"daycares"``・``"combination"``・ 

125 ``"output"`` をキーに、``{内部列名: 元ファイルの列名}`` の dict を持つ。 

126 既定値は ``chilmai.generic.config.DEFAULT_CONFIG``。 

127 combination_file_bytes: 組み合わせファイルの内容。任意。 

128 combination_file_format: 組み合わせファイルの形式。任意。 

129 ``combination_file_bytes`` と両方指定したときのみ組み合わせを検証する。 

130 

131 Returns: 

132 以下のキーを持つ dict: 

133 

134 - ``is_valid``（bool）: エラーがなければ ``True``。 

135 - ``errors``（list[dict]）: エラーのリスト。各要素は 

136 ``{"message": str, "type": str, "code": int | None}``。 

137 - ``warnings``（list[str]）: 警告メッセージのリスト。 

138 - ``summary``（dict）: ``children_count``・``daycares_count`` の件数集計。 

139 

140 自治体カスタムバリデーションまたは組み合わせファイルの検証で 

141 エラーが出た場合は、その時点で ``is_valid=False`` を返し、 

142 ChilmAI 汎用バリデーションは実行しない。 

143 

144 Raises: 

145 ChilmError: ファイルのパースに失敗した場合（形式不正、必要な列がないなど）。 

146 """ 

147 children_df = self.parser.parse_children( 

148 file_bytes=children_file_bytes, 

149 file_format=children_file_format, 

150 mapping=mapping.get("children", {}), 

151 ) 

152 daycares_df = self.parser.parse_daycares( 

153 file_bytes=daycares_file_bytes, 

154 file_format=daycares_file_format, 

155 mapping=mapping.get("daycares", {}), 

156 ) 

157 custom_errors = self.preprocessor.validate(children_df, daycares_df) 

158 if custom_errors: 

159 return { 

160 "is_valid": False, 

161 "errors": custom_errors, 

162 "warnings": [], 

163 "summary": { 

164 "children_count": int(len(children_df)), 

165 "daycares_count": int(len(daycares_df)), 

166 }, 

167 } 

168 children_df = self.preprocessor.transform_children(children_df) 

169 daycares_df = self.preprocessor.transform_daycares(daycares_df) 

170 

171 combination_df: pd.DataFrame | None = None 

172 if combination_file_bytes and combination_file_format: 

173 combination_df, combo_errors = self._parse_combination( 

174 combination_file_bytes, 

175 combination_file_format, 

176 mapping, 

177 children_df, 

178 daycares_df, 

179 ) 

180 if combo_errors: 

181 return { 

182 "is_valid": False, 

183 "errors": combo_errors, 

184 "warnings": [], 

185 "summary": { 

186 "children_count": int(len(children_df)), 

187 "daycares_count": int(len(daycares_df)), 

188 }, 

189 } 

190 

191 return self.validator.validate( 

192 children_df, 

193 daycares_df, 

194 children_mapping=mapping.get("children", {}), 

195 daycares_mapping=mapping.get("daycares", {}), 

196 combination_df=combination_df, 

197 )

199 def match( 

200 self, 

201 *, 

202 children_file_bytes: bytes, 

203 children_file_format: str, 

204 daycares_file_bytes: bytes, 

205 daycares_file_format: str, 

206 mapping: dict[str, dict[str, str]], 

207 solver_config: dict[str, Any] | None = None, 

208 combination_file_bytes: bytes | None = None, 

209 combination_file_format: str | None = None, 

210 ) -> dict[str, Any]: 

211 """アップロードされたファイルを検証し、CP-SAT マッチングを実行する。 

212 

213 公開 ``/match`` HTTP エンドポイントと同じ中核結果に加え、 

214 Web UI の Excel 出力で使う出力列・行データを返す。すべてキーワード引数で渡す。 

215 

216 Args: 

217 children_file_bytes: 申込者ファイルの内容。 

218 children_file_format: 申込者ファイルの形式。``"csv"`` または ``"xlsx"``。 

219 daycares_file_bytes: 保育所ファイルの内容。 

220 daycares_file_format: 保育所ファイルの形式。``"csv"`` または ``"xlsx"``。 

221 mapping: 列名マッピング。``"children"``・``"daycares"``・``"combination"``・ 

222 ``"output"`` をキーに、``{内部列名: 元ファイルの列名}`` の dict を持つ。 

223 既定値は ``chilmai.generic.config.DEFAULT_CONFIG``。 

224 solver_config: ソルバー設定。``max_time_seconds``（float、既定 ``10.0``）で 

225 CP-SAT の打ち切り時間を指定する。省略時は既定値を使う。 

226 combination_file_bytes: 組み合わせファイルの内容。任意。 

227 combination_file_format: 組み合わせファイルの形式。任意。 

228 ``combination_file_bytes`` と両方指定したときのみ組み合わせを使う。 

229 

230 Returns: 

231 以下のキーを持つ dict: 

232 

233 - ``matching_result_dict``（dict[str, str | None]）: 申込者 ID → 内定した 

234 保育所 ID。不成立の申込者は ``None``。 

235 - ``household_result_dict``（dict）: 世帯 ID → その世帯の 

236 ``child_ids``・``assigned``・``selected_combo`` など。 

237 - ``daycare_name_dict``（dict[str, str]）: 保育所 ID → 保育所名。 

238 - ``matched_children``（dict）: 成立件数の集計。``total``・``only_child``・ 

239 ``siblings``・``applied_total``・``by_age``（年齢別の内訳）。 

240 - ``transfer_back_count``（int）: 在籍中の保育所に再度内定した件数。 

241 該当がある場合のみ含まれる。 

242 ``mapping["output"]["exclude_transfer_back"] == "true"`` のときは 

243 該当申込者の結果を空欄に落とし、``matched_children`` からも差し引く。 

244 - ``meta``（dict）: ``algorithm``・``is_optimal`` などの実行情報。 

245 - ``output_columns``（list[str]）: 出力ファイルの列名リスト 

246 （元の申込ファイル全列 + 結果列）。 

247 - ``output_rows``（list[dict]）: 出力ファイルの行データ。 

248 ``BasePreprocessor.transform_output`` の加工後の内容。 

249 

250 Raises: 

251 ValueError: バリデーションエラーで中断した場合。第 2 引数に 

252 ``validate()`` と同じ構造の dict（``is_valid``・``errors``・ 

253 ``warnings``・``summary``）が入る。 

254 ChilmError: ファイルのパースに失敗した場合、申込者ファイルに 

255 申請者 ID 列が見つからない場合、出力列名の設定が重複している場合、 

256 出力列名が申込者ファイルの既存列と衝突する場合。 

257 """ 

258 children_df = self.parser.parse_children( 

259 file_bytes=children_file_bytes, 

260 file_format=children_file_format, 

261 mapping=mapping.get("children", {}), 

262 ) 

263 daycares_df = self.parser.parse_daycares( 

264 file_bytes=daycares_file_bytes, 

265 file_format=daycares_file_format, 

266 mapping=mapping.get("daycares", {}), 

267 ) 

268 custom_errors = self.preprocessor.validate(children_df, daycares_df) 

269 if custom_errors: 

270 raise ValueError( 

271 "Validation failed", 

272 { 

273 "is_valid": False, 

274 "errors": custom_errors, 

275 "warnings": [], 

276 "summary": { 

277 "children_count": int(len(children_df)), 

278 "daycares_count": int(len(daycares_df)), 

279 }, 

280 }, 

281 ) 

282 children_df = self.preprocessor.transform_children(children_df) 

283 daycares_df = self.preprocessor.transform_daycares(daycares_df) 

284 

285 combination_df: pd.DataFrame | None = None 

286 if combination_file_bytes and combination_file_format: 

287 combination_df, combo_errors = self._parse_combination( 

288 combination_file_bytes, 

289 combination_file_format, 

290 mapping, 

291 children_df, 

292 daycares_df, 

293 ) 

294 if combo_errors: 

295 raise ValueError( 

296 "Validation failed", 

297 { 

298 "is_valid": False, 

299 "errors": combo_errors, 

300 "warnings": [], 

301 "summary": { 

302 "children_count": int(len(children_df)), 

303 "daycares_count": int(len(daycares_df)), 

304 }, 

305 }, 

306 ) 

307 

308 validation = self.validator.validate( 

309 children_df, 

310 daycares_df, 

311 children_mapping=mapping.get("children", {}), 

312 daycares_mapping=mapping.get("daycares", {}), 

313 combination_df=combination_df, 

314 ) 

315 if not validation["is_valid"]: 

316 raise ValueError("Validation failed", validation) 

317 config = solver_config or {} 

318 max_time_seconds = float(config.get("max_time_seconds", 10.0)) 

319 result = self.matcher.match( 

320 children_df, 

321 daycares_df, 

322 max_time_seconds=max_time_seconds, 

323 combination_df=combination_df, 

324 ) 

325 

326 original_df = self.parser.read_raw(file_bytes=children_file_bytes, file_format=children_file_format) 

327 child_id_col = mapping.get("children", {}).get("child_id", "child_id") 

328 output_config = mapping.get("output", {}) 

329 result_id_col = output_config.get("result_daycare_id", "入所選考結果保育所ID") 

330 result_name_col = output_config.get("result_daycare_name", "入所選考結果保育所名") 

331 

332 child_id_col = _resolve_original_col(child_id_col, original_df) 

333 if child_id_col not in original_df.columns: 

334 if "child_id" in original_df.columns: 

335 child_id_col = "child_id" 

336 else: 

337 raise ChilmError( 

338 f"申込者ファイルに「{child_id_col}」列が見つかりません。" 

339 "設定画面で「申請者ID」の列名を確認してください。", 

340 code=ErrorCode.CHILD_ID_COL_NOT_FOUND, 

341 ) 

342 

343 if result_id_col == result_name_col: 

344 raise ChilmError( 

345 f"出力列名の設定が重複しています: '{result_id_col}'" 

346 "「入所選考結果保育所ID」と「入所選考結果保育所名」に別々の列名を設定してください。", 

347 code=ErrorCode.DUPLICATE_OUTPUT_COL_NAMES, 

348 ) 

349 norm_original_cols = {unicodedata.normalize("NFKC", str(c)) for c in original_df.columns} 

350 conflicts = [ 

351 col 

352 for col in (result_id_col, result_name_col) 

353 if unicodedata.normalize("NFKC", col) in norm_original_cols 

354 ] 

355 if conflicts: 

356 raise ChilmError( 

357 f"出力列名が申込者ファイルの既存列と重複しています: {conflicts}" 

358 "設定画面で別の列名に変更してください。", 

359 code=ErrorCode.OUTPUT_COL_CONFLICTS_INPUT, 

360 ) 

361 

362 matching_result_dict: dict[str, str | None] = result["matching_result_dict"] 

363 daycare_name_dict: dict[str, str] = result.get("daycare_name_dict", {}) 

364 

365 child_id_series = original_df[child_id_col].map(lambda v: norm_id(str(v))) 

366 result_id_series = child_id_series.map(matching_result_dict).fillna("") 

367 result_name_series = result_id_series.map(daycare_name_dict).fillna("") 

368 

369 enrolled_id_col = mapping.get("children", {}).get("enrolled_daycare_id", "在籍保育所ID") 

370 enrolled_id_col = _resolve_original_col(enrolled_id_col, original_df) 

371 if enrolled_id_col not in original_df.columns and "enrolled_daycare_id" in original_df.columns: 

372 enrolled_id_col = "enrolled_daycare_id" 

373 if enrolled_id_col in original_df.columns: 

374 enrolled_id_series = original_df[enrolled_id_col].map( 

375 lambda v: "" if pd.isna(v) else norm_id(str(v)) 

376 ) 

377 same_daycare_mask = (result_id_series != "") & (result_id_series == enrolled_id_series) 

378 blanked_count = int(same_daycare_mask.sum()) 

379 if blanked_count > 0: 

380 blanked_ids = set(child_id_series[same_daycare_mask].tolist()) 

381 

382 blanked_only = 0 

383 blanked_siblings = 0 

384 for hh in result.get("household_result_dict", {}).values(): 

385 hh_children = hh.get("child_ids", []) 

386 blanked_in_hh = sum(1 for cid in hh_children if cid in blanked_ids) 

387 if blanked_in_hh == 0: 

388 continue 

389 # matcher は世帯人数でバケットを決めるので、 

390 # きょうだい世帯で一部の子だけブランクされる場合も siblings として補正する。 

391 if len(hh_children) == 1: 

392 blanked_only += blanked_in_hh 

393 else: 

394 blanked_siblings += blanked_in_hh 

395 

396 age_lookup = dict( 

397 zip( 

398 children_df["child_id"].map(lambda v: norm_id(str(v))), 

399 children_df["age"].map(lambda v: str(int(v))), 

400 ) 

401 ) 

402 blanked_by_age: dict[str, int] = {} 

403 for cid in blanked_ids: 

404 age_key = age_lookup.get(cid) 

405 if age_key is not None: 

406 blanked_by_age[age_key] = blanked_by_age.get(age_key, 0) + 1 

407 

408 mc = result.get("matched_children", {}) 

409 by_age = {age: {**bucket} for age, bucket in mc.get("by_age", {}).items()} 

410 for age_key, blanked in blanked_by_age.items(): 

411 if age_key in by_age: 

412 by_age[age_key]["matched"] = max( 

413 0, 

414 by_age[age_key].get("matched", 0) - blanked, 

415 ) 

416 result["matched_children"] = { 

417 **mc, 

418 "total": mc.get("total", 0) - blanked_count, 

419 "only_child": mc.get("only_child", 0) - blanked_only, 

420 "siblings": mc.get("siblings", 0) - blanked_siblings, 

421 "by_age": by_age, 

422 } 

423 result["transfer_back_count"] = blanked_count 

424 

425 if output_config.get("exclude_transfer_back", "false") == "true": 

426 result_id_series = result_id_series.where(~same_daycare_mask, other="") 

427 result_name_series = result_name_series.where(~same_daycare_mask, other="") 

428 result["matching_result_dict"] = { 

429 cid: (None if cid in blanked_ids else did) 

430 for cid, did in result["matching_result_dict"].items() 

431 } 

432 result["household_result_dict"] = { 

433 hh_id: { 

434 **hh, 

435 "assigned": [ 

436 None if cid in blanked_ids else d 

437 for cid, d in zip(hh.get("child_ids", []), hh.get("assigned", [])) 

438 ], 

439 "selected_combo": [ 

440 None if cid in blanked_ids else d 

441 for cid, d in zip(hh.get("child_ids", []), hh.get("selected_combo", [])) 

442 ], 

443 } 

444 for hh_id, hh in result.get("household_result_dict", {}).items() 

445 } 

446 

447 output_df = original_df.copy() 

448 output_df[result_id_col] = result_id_series 

449 output_df[result_name_col] = result_name_series 

450 

451 output_df = self.preprocessor.transform_output(output_df, result) 

452 

453 result["output_columns"] = list(output_df.columns) 

454 result["output_rows"] = output_df.fillna("").to_dict(orient="records") 

455 

456 return result
